### D.1 비교사례 보강 전용 타입

```ts
/** 실거래 API 원시 — 지번은 마스킹되어 옵니다 */
export interface CompRaw {
  sigungu: string;
  bjdong: string;
  jibunMasked: string;      // '1***'
  kind: '일반' | '집합';
  zoning: string;
  mainUse: string;
  builtYear: number;
  landSqm: number | null;   // 집합건물은 null
  buildingSqm: number;
  priceKrw: number;
  contractYm: string;
}

export interface CompEnrich {
  identify: string;                     // '당산동5가 대로변 근생'

  condition: string;                    // '정상' | '리모델링' | '공실'
}

export type Comp = CompRaw & { enrich?: CompEnrich };

/** 보강된 건만 비교표에 올립니다 */
export const usable = (c: Comp) => c.enrich != null;
```
