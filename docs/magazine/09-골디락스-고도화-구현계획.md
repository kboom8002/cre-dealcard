  try {
    const { error } = await resend.emails.send({
      from: `${payload.brokerName} <magazine@credeal.net>`,
      to: payload.to,
      subject: `[${payload.brokerName}] ${payload.magazineTitle}`,
      html: buildMagazineHtml(payload),  // 아래 템플릿 함수
    });
    if (error) { console.error('[Email] Send failed:', error); return false; }
    return true;
  } catch (err) {
    console.error('[Email] Unexpected error:', err);
    return false;
  }
}

export async function generatePersonalizedInsert(
  interestTags: { regions?: string[]; assetTypes?: string[]; topics?: string[] },
  editionContent: { ai_briefing: string; theme_title: string },
  recentNews: Array<{ title: string; source: string }>,

  if (!interestTags.regions?.length && !interestTags.topics?.length) {
    return ''; // 관심사 미설정 시 빈 문자열

// src/domain/magazine/buyer-temperature.ts

export type BuyerTemperature = '🔥 적극검토' | '📈 관심' | '⏸️ 관망' | '❄️ 냉각' | '⚪ 미확인';

export interface TemperatureConfig {
  label: BuyerTemperature;
  color: string;
  min: number;
}

export const TEMPERATURE_TIERS: TemperatureConfig[] = [
  { label: '🔥 적극검토', color: '#ef4444', min: 80 },
  { label: '📈 관심',     color: '#22c55e', min: 60 },
  { label: '⏸️ 관망',    color: '#f59e0b', min: 40 },
  { label: '❄️ 냉각',    color: '#6b7280', min: 20 },
  { label: '⚪ 미확인',   color: '#374151', min: 0 },
];

/**
 * engagement (subscriber-profile.ts computeEngagementScore, 0~100)와
 * crossChannel (cross-channel-score.ts calculateLeadScore, 0~100+)을
 * 가중 평균하여 매수 온도를 반환.
 *
 * 가중치: engagement 40% + crossChannel 60% (실제 행동 비중↑)
 */
export function getBuyerTemperature(
  engagement: number,
  crossChannel: number = 0,
): TemperatureConfig {
  const composite = Math.round(engagement * 0.4 + Math.min(crossChannel, 100) * 0.6);
  return TEMPERATURE_TIERS.find(t => composite >= t.min) || TEMPERATURE_TIERS[4];
}

import type { SupabaseClient } from '@supabase/supabase-js';

export interface MonthlyTransactionSummary {
  yearMonth: string;
  region: string;
  totalCount: number;
  avgPrice: number;
  maxPrice: number;
  minPrice: number;
  avgPricePerSqm: number;
  previousMonthCount: number;
  countChangePercent: number;
  byUsageType: Record<string, number>;
}

/**
 * external_transactions 테이블에서 월간 집계.
 * gov-premium-apis.ts가 매일 적재한 데이터를 활용.
 */
export async function summarizeMonthlyTransactions(
  supabase: SupabaseClient,
  yearMonth: string,  // 'YYYY-MM'
  regions: string[],
): Promise<MonthlyTransactionSummary[]> {
  const results: MonthlyTransactionSummary[] = [];
  const startDate = `${yearMonth}-01`;
  const endDate = `${yearMonth}-31`;

  // 전월 계산
  const [y, m] = yearMonth.split('-').map(Number);
  const prevMonth = m === 1 ? `${y - 1}-12` : `${y}-${String(m - 1).padStart(2, '0')}`;
  const prevStart = `${prevMonth}-01`;
  const prevEnd = `${prevMonth}-31`;

  for (const region of regions) {
    // 당월
    const { data: current } = await supabase
      .from('external_transactions')
      .select('transaction_price, building_area, usage_type')
      .eq('district', region)
      .gte('transaction_date', startDate)
      .lte('transaction_date', endDate);

    // 전월
    const { count: prevCount } = await supabase
      .from('external_transactions')
      .select('id', { count: 'exact', head: true })
      .eq('district', region)
      .gte('transaction_date', prevStart)
      .lte('transaction_date', prevEnd);

    if (!current || current.length === 0) continue;

    const prices = current.map((t: any) => t.transaction_price);
    const areas = current.filter((t: any) => t.building_area > 0);
    const byUsage: Record<string, number> = {};
    current.forEach((t: any) => { byUsage[t.usage_type] = (byUsage[t.usage_type] || 0) + 1; });

    results.push({
      yearMonth,
      region,
      totalCount: current.length,
      avgPrice: Math.round(prices.reduce((a: number, b: number) => a + b, 0) / prices.length),
      maxPrice: Math.max(...prices),
      minPrice: Math.min(...prices),
      avgPricePerSqm: areas.length > 0
        ? Math.round(areas.reduce((s: number, t: any) => s + t.transaction_price / t.building_area, 0) / areas.length)
        : 0,
      previousMonthCount: prevCount || 0,
      countChangePercent: prevCount ? Math.round(((current.length - prevCount) / prevCount) * 100) : 0,
      byUsageType: byUsage,
    });
  }

  return results;
}
